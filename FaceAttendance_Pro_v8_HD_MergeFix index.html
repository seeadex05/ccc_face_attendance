📖 Setup Instructions

1. Open Google Sheets → Create a new blank spreadsheet.

2. Go to Extensions → Apps Script.

3. Paste the provided Apps Script code (see file).

4. Click Deploy → New deployment.

5. Choose type: Web app.

6. Set "Who has access" to Anyone.

7. Copy the Web App URL and paste it above.

8. Click Save URL, then Push to Google Sheet.

async function runMobileDiagnostics() { const el = document.getElementById('mobileDiagnostic'); if (!el) return; const rows = []; rows.push(`Browser: ${navigator.userAgent.includes('Chrome') ? 'Chrome' : navigator.userAgent.includes('Safari') ? 'Safari' : 'Other'}`); rows.push(`HTTPS / secure context: ${window.isSecureContext ? '✅ YES' : '❌ NO'}`); rows.push(`Camera API: ${navigator.mediaDevices?.getUserMedia ? '✅ Available' : '❌ Not available'}`); if (navigator.permissions?.query) { try { const p = await navigator.permissions.query({ name: 'camera' }); rows.push(`Camera permission: ${p.state === 'granted' ? '✅ Allowed' : p.state === 'prompt' ? '🟡 Ask when camera opens' : '❌ Blocked'}`); } catch (_) { rows.push('Camera permission: Browser did not expose permission state'); } } rows.push(`Face models: ${modelsLoaded ? '✅ Loaded' : '❌ Not loaded'}`); rows.push(`Registered faces: ${DB.people.length}`); rows.push(`Google Apps Script URL: ${DB.scriptUrl ? '✅ Saved' : '❌ Not configured'}`); if (DB.scriptUrl) { try { const r = await fetch(DB.scriptUrl + '?action=load&check=' + Date.now(), { cache: 'no-store' }); if (!r.ok) throw new Error('HTTP ' + r.status); const d = await r.json(); rows.push(`Google API: ${d.error ? '❌ '+d.error : '✅ Connected'}`); } catch (e) { rows.push(`Google API: ❌ ${e.message}`); } } el.innerHTML = rows.map(x => `
${x}
`).join(''); } window.addEventListener('load', () => setTimeout(runMobileDiagnostics, 1000));
